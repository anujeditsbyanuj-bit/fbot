# eporner_downloader.py backend — the real, actively-maintained yt-dlp
# package (has a dedicated, well-tested Eporner extractor). Far more
# solid than porn_fetch_downloader's route for this same site: those
# eaf_base_api/unofficial-api-for-* packages below are unverified
# GitHub-only projects this environment couldn't even pip-install to
# check against, per porn_fetch_downloader.py's own docstring.
#
# [default] extra + curl_cffi below mirror the reference "src" project's
# own ytdl.py setup: [default] pulls in yt-dlp's bundled JS-challenge
# solver dependencies, and curl_cffi backs yt-dlp's browser-TLS
# impersonation (used as a Cloudflare-anti-bot fallback — see
# eporner_downloader.py's _impersonate_available()). Pinned to a recent
# release for the same reason src's requirements.txt pins it: an
# unpinned "yt-dlp" can silently stay on a stale cached build that no
# longer matches a site's current page structure.
yt-dlp[default,curl-cffi]

# JS runtime yt-dlp needs to solve YouTube's challenge (required since
# yt-dlp 2025.11.12) — installed as a pip package specifically because
# render.yaml's buildCommand is just "pip install -r requirements.txt"
# (no shell access to run Deno's own curl-based installer there), even
# though the Dockerfile path also installs it via apt/curl for Docker-
# based deploys.
deno
curl_cffi>=0.10,<0.14

# PO-token support for YouTube (see pot_provider.py + ytdlp_downloader.py's
# player_client logic) — this is the yt-dlp-side plugin that talks to the
# local bgutil-ytdlp-pot-provider HTTP server pot_provider.py starts at
# boot. Safe to have installed even if that server never comes up on this
# host (e.g. missing native build tools for its "canvas" dependency) — the
# plugin just won't find a provider to use and yt-dlp falls back exactly
# as if this weren't installed at all.
bgutil-ytdlp-pot-provider
